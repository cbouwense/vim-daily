title: Files and buffers: :e :ls :b Ctrl-^
keys: :e  :ls  :b  :bn  :bp  Ctrl-^  :wa
require: :e\s :: :e
require: :b\s?\S :: :b
require: <C-\^> :: Ctrl-^
require: :ls<CR> :: :ls
ext: ts

Each open file is a buffer. Vim keeps many open; a window shows one of them.

  :e user-service.ts   open (edit) a file; Tab completes the name
  :ls                  list the open buffers
  :b name              switch by number or by part of the name
  :bn  :bp             next / previous buffer
  Ctrl-^               back to the buffer you were just in (Ctrl-6)
  :w                   save this buffer         :wa   save all of them

Vim will not leave a buffer with unsaved changes (error E37): :w first.
Finish today with :wqa, which saves every buffer and quits.

== Rename applyDiscount to applyCoupon: once in discount.ts, twice in
== checkout.ts. :w before each switch, :ls to look, ":b disc" to go back.
@ file: discount.ts
export function applyDiscount(total: number, code: string) {
  return code === "SAVE10" ? total * 0.9 : total;
}
@ file: checkout.ts
import { applyDiscount } from "./discount";

export function checkout(total: number, code: string) {
  return applyDiscount(total, code);
}
