title: Split windows: :sp :vsp Ctrl-w
keys: :sp  :vsp  Ctrl-w h/j/k/l  Ctrl-w c  Ctrl-w o
require: :vsp :: :vsp
require: :sp :: :sp
require: <C-w>[hjklwp] :: Ctrl-w h/j/k/l
require: <C-w>[co] :: Ctrl-w c
ext: ts

Split the screen to see two files, or two parts of one file, at once:

  :sp file         split, file on top        :vsp file    split, file on the left
  Ctrl-w h/j/k/l   move to the window left / below / above / right
  Ctrl-w w         cycle through the windows
  Ctrl-w c         close this window         Ctrl-w o     keep only this one

With no file name, :sp and :vsp show the same buffer twice. This lesson pane
is a window too: Ctrl-w k can land here, Ctrl-w j goes back down, and
:Lesson reopens it after Ctrl-w o. Closing a window does not close the file.

== Open settings-form.ts and type :sp to see it twice. Rename MAX_LEN to
== MAX_LENGTH at the top; Ctrl-w j, G, and rename it at the bottom too.
== Close the extra window with Ctrl-w c.
@ file: settings-form.ts
export const MAX_LEN = 40;

export type Field = {
  name: string;
  label: string;
  value: string;
  required: boolean;
};

export function emptyField(name: string, label: string): Field {
  return { name, label, value: "", required: false };
}

export function isFilled(field: Field) {
  return field.value.trim().length > 0;
}

export function missing(fields: Field[]) {
  return fields.filter((f) => f.required && !isFilled(f));
}

export function labelFor(field: Field) {
  return field.required ? field.label + " *" : field.label;
}

export function tooLong(field: Field) {
  return field.value.length > MAX_LEN;
}

-- :sp<CR>
-- /MAX_LEN<CR>
-- cwMAX_LENGTH<Esc>

-- N

-- <C-w>c
