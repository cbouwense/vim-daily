title: Substitute: :s and :%s
keys: :s/old/new/  :s/old/new/g  :5,9s/old/new/g  :%s/old/new/gc
require: :s/ :: :s
require: s/[^/]*/[^/]*/c?gc?<CR> :: the g flag
require: :[^:/]+,[^:/]+s/ :: a line range, as in :12,15s/old/new/
require: :%s/ :: :%s
require: s/[^/]*/[^/]*/g?cg?<CR> :: the c flag
ext: ts

:s replaces text. The old and the new text go between slashes:

  :s/old/new/          the first match on the cursor line
  :s/old/new/g         every match on the line (g for global)
  :12,15s/old/new/g    every match on lines 12 to 15
  :%s/old/new/g        every match in the file (% means all lines)
  :%s/old/new/gc       the same, but ask at each match: y yes, n no, q quit

The old text is a pattern like the one / searches for, so . and * are
special in it. One u undoes a whole substitute.

== The three seller lines were pasted from the buyer line. Replace every
== buyer with seller on just those lines:  :{L+2},{L+4}s/buyer/seller/g
function partyLines(buyer: Party, seller: Party) {
  const buyerLine = `${buyer.name} <${buyer.email}>`;
  const sellerLine = `${buyer.name} <${buyer.email}>`;
  const sellerCity = buyer.address?.city ?? buyer.region;
  const sellerPhone = buyer.phone ?? buyer.mobile;
  return { buyerLine, sellerLine, sellerCity, sellerPhone };
}
-- :{L+2},{L+4}s/buyer/seller/g<CR>

== Save with :w, open invoice-totals.ts with :e, and rename amt to amount in
== the whole file:  :%s/amt/amount/g
@ file: invoice-totals.ts
export type InvoiceLine = { sku: string; amt: number; vatRate: number };

export function invoiceTotal(lines: InvoiceLine[]): number {
  let amt = 0;
  for (const line of lines) {
    amt += line.amt * (1 + line.vatRate);
  }
  return amt;
}
